1. **Add verifyCitations to chatbot/src/lib/citations.ts:**

Add a new exported function below the existing parseCitations:

```typescript
export interface VerifiedCitation extends Citation {
  verified: boolean;
}

export function verifyCitations(
  citations: Citation[],
  availableToolNames: string[]
): VerifiedCitation[] {
  // Normalize available tool names to lowercase for comparison
  const normalizedNames = new Set(availableToolNames.map(n => n.toLowerCase()));

  return citations.map(citation => {
    // Citations without a tool reference are considered verified (generic citations)
    if (!citation.tool) {
      return { ...citation, verified: true };
    }
    // Check if the cited tool exists in the available tools
    const verified = normalizedNames.has(citation.tool.toLowerCase());
    if (!verified) {
      console.warn(`Unverified citation: tool "${citation.tool}" not found in content repository`);
    }
    return { ...citation, verified };
  });
}
```

The function takes the flat list of tool names (not ToolData objects) so it has no dependency on content-types and is easy to test. The chat route will extract tool names from the ContentContext.

export async function POST(req: NextRequest) {

  sessionStore.addMessage(sessionId, { role: 'user', content: message });

  try {
    // NEW: Load relevant content for this message
    const { tools } = findRelevantContent(message);
    const contentContext = formatContentForPrompt(tools);

    const provider = getLLMProvider();
    const messages = sessionStore.getMessages(sessionId);
    // CHANGED: Pass contentContext to provider
    const response = await provider.chat({
      messages,
      accessMode: session.accessMode,
      contentContext: contentContext || undefined,
    });

    sessionStore.addMessage(sessionId, {
      role: 'assistant',
      content: response.content,
    });

    const { cleanContent, citations } = parseCitations(response.content);

    // NEW: Verify citations against available tool names
    const toolNames = tools.map(t => t.name);
    const verifiedCitations = verifyCitations(citations, toolNames);

    return NextResponse.json({
      message: cleanContent,
      citations: verifiedCitations,
    });
  } catch (error) {
    // ... existing error handling stays the same
  }
}
