       ```typescript
       import { spring, interpolate, useCurrentFrame, useVideoConfig } from 'remotion';

       export function useSpringFadeIn(delay: number = 0) {
         const frame = useCurrentFrame();
         const { fps } = useVideoConfig();
         const progress = spring({ frame: frame - delay, fps, config: { damping: 20, mass: 0.5 } });
         return { opacity: progress, transform: `translateY(${interpolate(progress, [0, 1], [20, 0])}px)` };
       }

       export function useSpringScale(delay: number = 0) {
         const frame = useCurrentFrame();
         const { fps } = useVideoConfig();
         const progress = spring({ frame: frame - delay, fps, config: { damping: 15, stiffness: 100 } });
         return { transform: `scale(${interpolate(progress, [0, 1], [0.8, 1])})`, opacity: progress };
       }
       ```

    The script exports these testable helpers:
    ```typescript
    export const SCENES = [
      { name: 'typography', startFrame: 0, endFrame: 89 },
      { name: 'brand-elements', startFrame: 90, endFrame: 179 },
      { name: 'motion-test', startFrame: 180, endFrame: 299 },
    ] as const;

    export function getScreenshotFrames(scene: typeof SCENES[number]): number[] {
      const start = scene.startFrame;
      const end = scene.endFrame;
      const mid = Math.floor((start + end) / 2);
      return [start, mid, end];
    }

    export function getScreenshotPath(sceneName: string, frame: number): string {
      return path.join(process.cwd(), `output/verification/screenshots/${sceneName}-frame-${frame}.png`);
    }
    ```
