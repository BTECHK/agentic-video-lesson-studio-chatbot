```typescript
import { NextRequest, NextResponse } from 'next/server';
import { sessionStore } from '@/lib/session';
import type { AccessMode } from '@/lib/types';

export async function POST(req: NextRequest) {
  const body = await req.json();
  const accessMode = (body.accessMode === 'student' ? 'student' : 'demo') as AccessMode;
  const session = sessionStore.create(accessMode);
  return NextResponse.json({ sessionId: session.id, accessMode: session.accessMode });
}
```

2. Create `chatbot/src/app/api/chat/route.ts`:

This is the core wiring. It:
- Accepts POST with `{ message, sessionId, accessMode }` body
- Validates input with `ChatRequestSchema` (Zod)
- Retrieves session from `sessionStore.get(sessionId)` -- returns 404 if expired
- Adds user message to session via `sessionStore.addMessage()`
- Builds LLMRequest with full message history from `sessionStore.getMessages()`
- Calls `getLLMProvider().chat(request)` to get LLM response
- Adds assistant message to session
- Parses citation markers from response content -- regex pattern: `\[Source: ([^\]]+)\]`
- Returns `{ message, citations }` as ChatResponse

Citation parsing logic:
```typescript
function parseCitations(content: string): { cleanContent: string; citations: Citation[] } {
  const citationRegex = /\[Source: ([^\]]+)\]/g;
  const citations: Citation[] = [];
  let match;
  let index = 0;

  while ((match = citationRegex.exec(content)) !== null) {
    const raw = match[1];
    // Parse "Module X, Lesson Y" or "General AI knowledge"
    const moduleMatch = raw.match(/Module\s+(\d+)/i);
    const lessonMatch = raw.match(/Lesson\s+(\d+)/i);
    citations.push({
      id: `cite-${index++}`,
      module: moduleMatch ? `module-${moduleMatch[1]}` : 'general',
      lesson: lessonMatch ? `lesson-${lessonMatch[1]}` : 'general',
      snippet: raw,
    });
  }

  return { cleanContent: content, citations };
}
```

Error handling:
- Invalid input (Zod validation fails) -> 400 with error details
- Session not found -> 404 with "Session expired or not found"
- LLM call fails -> 500 with "Failed to generate response"
- All errors return JSON `{ error: string }`

| Boundary | Description |
|----------|-------------|
| Client -> /api/chat | Untrusted user messages sent to LLM |
| Client -> /api/session | Session creation from untrusted client |
| Server -> Anthropic API | Forwarding user content to external LLM |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-08 | Injection | /api/chat user message | mitigate | Zod validation limits message to 2000 chars; system prompt is server-side only; user content is always role:'user' |
| T-05-09 | Denial of Service | /api/chat rate abuse | accept | No rate limiting in v1; session-based usage provides natural throttling; add rate limiting if abuse observed |
| T-05-10 | Repudiation | Chat history | accept | In-memory sessions expire; no persistent logging of conversations in v1; acceptable for bootstrap |
| T-05-11 | Information Disclosure | LLM hallucination | mitigate | System prompt instructs course-first behavior with explicit source citation; bootstrap mode labels responses as "General AI knowledge" |
</threat_model>
