Implement access control per D-01 (dual-mode), D-02 (dev bypass), D-03 (demo module-scoped).

1. Create `chatbot/src/lib/access.ts`:
```typescript
import type { AccessMode } from './types';

export function getAccessMode(): AccessMode {
  // Default is demo mode -- no auth required (per D-01)
  return 'demo';
}

export function validateAccessCode(code: string): boolean {
  // Dev bypass per D-02
  if (process.env.NEXT_PUBLIC_DEV_BYPASS === 'true') {
    return true;
  }
  const validCode = process.env.ACCESS_CODE || 'demo-access-code';
  return code === validCode;
}

export function getDemoModuleScope(): string {
  // Per D-03: demo content is module-scoped and configurable
  return process.env.DEMO_MODULE_SCOPE || 'module-1';
}

export function isModuleAccessible(moduleId: string, mode: AccessMode): boolean {
  if (mode === 'student') return true; // Students get full access per D-03
  const demoScope = getDemoModuleScope();
  return moduleId === demoScope;
}
```

2. Create `chatbot/src/app/api/access/route.ts`:
```typescript
import { NextRequest, NextResponse } from 'next/server';
import { validateAccessCode } from '@/lib/access';

export async function POST(req: NextRequest) {
  const body = await req.json();
  const { code } = body as { code?: string };

  if (!code || typeof code !== 'string') {
    return NextResponse.json({ valid: false, error: 'Access code required' }, { status: 400 });
  }

  const valid = validateAccessCode(code);
  if (valid) {
    return NextResponse.json({ valid: true, mode: 'student' as const });
  }
  return NextResponse.json({ valid: false, error: 'Invalid access code' }, { status: 401 });
}
```

This implements:
- D-01: Demo mode is default (no auth). Student mode requires POST to /api/access with valid code.
- D-02: When NEXT_PUBLIC_DEV_BYPASS=true, any code is accepted.
- D-03: getDemoModuleScope() returns configurable module scope (defaults to "module-1"). isModuleAccessible() gates content by mode.
